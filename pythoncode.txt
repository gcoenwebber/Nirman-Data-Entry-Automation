import os
import json
import pandas as pd
from PIL import Image
from google import genai
from google.genai import types

# Initialize the Gemini Client
client = genai.Client(api_key="YOUR_GEMINI_API_KEY")

def extract_ratings_free(image_path):
    print(f"Processing {image_path} with Gemini...")
    
    # Open the image using PIL
    try:
        img = Image.open(image_path)
    except Exception as e:
        print(f"Could not open image {image_path}: {e}")
        return None

    prompt = """
    You are an expert data entry assistant. Look closely at this handwritten workshop feedback form image.
    Extract the scores/ratings for the following categories:
    1. Content Quality
    2. Trainer Performance
    3. Venue/Facilities

    The ratings should be integers on a 1-5 scale. If a rating is completely illegible, write "Review Needed".
    
    Respond STRICTLY in a clean JSON format like this:
    {
        "Content Quality": 5,
        "Trainer Performance": 4,
        "Venue/Facilities": 3
    }
    """

    # Call Gemini Flash (Vision capable and completely free in AI Studio)
    response = client.models.generate_content(
        model='gemini-2.5-flash', # Or 'gemini-1.5-flash' depending on current availability
        contents=[img, prompt],
        config=types.GenerateContentConfig(
            response_mime_type="application/json", # Forces Gemini to return pure JSON
            temperature=0.0
        ),
    )
    
    try:
        # Parse the JSON response
        data = json.loads(response.text.strip())
        data["File Name"] = os.path.basename(image_path)
        return data
    except Exception as e:
        print(f"Error parsing JSON response: {e}")
        return {"File Name": os.path.basename(image_path), "Content Quality": "Error", "Trainer Performance": "Error", "Venue/Facilities": "Error"}

# --- Execution ---
# Add your scanned form image filenames here
image_files = ["form1.jpg"] 
all_results = []

for image in image_files:
    result = extract_ratings_free(image)
    if result:
        all_results.append(result)

if all_results:
    # Convert to DataFrame and save to Excel
    df = pd.DataFrame(all_results)
    df = df[["File Name", "Content Quality", "Trainer Performance", "Venue/Facilities"]]
    df.to_excel("workshop_ratings_free.xlsx", index=False)
    print("\n Success! Check 'workshop_ratings_free.xlsx'")
